async function api(method: string, path: string, body?: object): Promise<any> {
  return new Promise((resolve, reject) => {
    const call = request({ hostname: "127.0.0.1", port, method, path, headers: { origin: `http://127.0.0.1:${port}`, ...(body ? { "content-type": "application/json" } : {}) } }, (response) => {
      let data = "";
      response.on("data", (chunk) => { data += chunk; });
      response.on("end", () => {
        if ((response.statusCode ?? 500) >= 400) reject(new Error(`${method} ${path} returned ${response.statusCode}: ${data}`));
        else resolve(JSON.parse(data));
      });
    });
    call.on("error", reject);
    call.end(body ? JSON.stringify(body) : undefined);
  });
}
